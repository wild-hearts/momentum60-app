Disallow: /team
Disallow: /insights
Disallow: /rules
Disallow: /summary
Disallow: /settings
Disallow: /auth
Disallow: /update-password

# Build helper that ships inside public/ and is therefore served.
Disallow: /downloads/convert.py

# The 60 daily anthem MP3s are roughly 200 MB in total and are streamed on
# demand, never precached. There is no reason for a crawler to pull them.
Disallow: /songs/
